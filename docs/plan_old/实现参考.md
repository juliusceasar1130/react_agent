## 2. RAG 中间件实现

### 2.1 完整实现代码

**文件**: `backend/app/agent/middleware/rag_middleware.py`

```python
# backend/app/agent/middleware/rag_middleware.py
"""
业务知识 RAG 中间件

在用户消息时自动检索业务知识，并将检索结果作为系统消息注入到 messages 中。
"""

from typing import Any, List, Optional
import logging

from langchain.agents.middleware import AgentMiddleware
from langchain_core.messages import BaseMessage, SystemMessage
from langchain_core.documents import Document

from backend.app.agent.state import CustomState

logger = logging.getLogger(__name__)


class BusinessRagMiddleware(AgentMiddleware[CustomState]):
    """
    业务知识 RAG 中间件

    - before_model: 在用户消息时检索业务知识，并将检索结果作为系统消息注入到 messages 中
    """

    state_schema = CustomState

    def __init__(self, vector_store, top_k: int = 5) -> None:
        """
        Args:
            vector_store: 业务向量库实例（需支持 .similarity_search(query, k) 方法）
            top_k: 检索的文档数量，默认 5
        """
        self.vector_store = vector_store
        self.top_k = top_k
        # 用于标记业务知识系统消息的标识
        self._rag_system_message_id = "__business_rag_context__"

    # --------- 工具方法 ---------

    @staticmethod
    def _is_human_message(msg: BaseMessage) -> bool:
        """判断是否为用户消息"""
        msg_type = getattr(msg, "type", None) or getattr(msg, "role", None)
        return msg_type == "human"

    @staticmethod
    def _format_knowledge_block(docs: List[Document]) -> str:
        """将检索结果格式化为系统提示词文本块"""
        if not docs:
            return ""

        parts = []
        for i, doc in enumerate(docs, start=1):
            meta = getattr(doc, "metadata", {}) or {}
            title = meta.get("title") or meta.get("source") or f"文档 #{i}"
            parts.append(f"### {title}\n\n{doc.page_content}")

        return "\n\n".join(parts)

    def _has_rag_system_message(self, messages: List[BaseMessage]) -> bool:
        """检查 messages 中是否已包含业务知识系统消息"""
        for msg in messages:
            if isinstance(msg, SystemMessage):
                # 检查是否有我们添加的业务知识标识
                content = getattr(msg, "content", "")
                if isinstance(content, str) and self._rag_system_message_id in content:
                    return True
                # 也检查 content_blocks 格式
                if hasattr(msg, "content_blocks"):
                    for block in msg.content_blocks:
                        if isinstance(block, dict) and block.get("type") == "text":
                            if self._rag_system_message_id in block.get("text", ""):
                                return True
        return False

    # --------- 钩子实现 ---------

    def before_model(self, state: CustomState) -> Optional[dict[str, Any]]:
        """
        在用户消息时检索业务知识，并将检索结果作为系统消息注入到 messages 中。
        """
        messages: List[BaseMessage] = state.get("messages", [])
        if not messages:
            return None

        last_msg = messages[-1]

        # 只在用户消息时执行检索
        if not self._is_human_message(last_msg):
            logger.debug("BusinessRagMiddleware.before_model: 非用户消息，跳过检索")
            return None

        user_query = getattr(last_msg, "content", None) or getattr(last_msg, "text", "")
        if not user_query:
            return None

        # 执行业务检索
        try:
            retrieved_docs: List[Document] = self.vector_store.similarity_search(
                user_query, k=self.top_k
            )
        except Exception as e:
            logger.error(f"BusinessRagMiddleware: 向量检索失败: {e}", exc_info=True)
            return None

        logger.info(f"BusinessRagMiddleware: 检索到 {len(retrieved_docs)} 条业务文档")

        # 格式化检索结果为系统提示词
        knowledge_block = self._format_knowledge_block(retrieved_docs)
        if not knowledge_block:
            logger.debug("BusinessRagMiddleware: 未检索到相关业务知识")
            return None

        # 构建业务知识系统消息内容
        rag_system_content = (
            f"{self._rag_system_message_id}\n\n"
            "## 业务知识库\n\n"
            "下面是与当前用户问题相关的业务资料，请在回答中充分利用这些信息：\n\n"
            f"{knowledge_block}\n"
        )

        # 创建系统消息
        rag_system_message = SystemMessage(content=rag_system_content)

        # 检查是否已经存在业务知识系统消息，如果存在则替换，否则添加到开头
        new_messages = []
        rag_message_added = False

        for msg in messages:
            if isinstance(msg, SystemMessage):
                content = getattr(msg, "content", "")
                # 如果找到已有的业务知识系统消息，替换它
                if isinstance(content, str) and self._rag_system_message_id in content:
                    if not rag_message_added:
                        new_messages.append(rag_system_message)
                        rag_message_added = True
                    # 跳过旧的消息
                    continue
                # 检查 content_blocks 格式
                elif hasattr(msg, "content_blocks"):
                    should_replace = False
                    for block in msg.content_blocks:
                        if isinstance(block, dict) and block.get("type") == "text":
                            if self._rag_system_message_id in block.get("text", ""):
                                should_replace = True
                                break
                    if should_replace:
                        if not rag_message_added:
                            new_messages.append(rag_system_message)
                            rag_message_added = True
                        continue

            new_messages.append(msg)

        # 如果没有找到已有的业务知识系统消息，则在开头添加
        if not rag_message_added:
            new_messages.insert(0, rag_system_message)

        logger.debug("BusinessRagMiddleware: 已将业务知识注入到 messages 作为系统消息")

        # 返回更新后的 state
        return {
            "messages": new_messages,
            "rag_context": retrieved_docs,
            "rag_query": user_query,
        }
```

## 3. 向量存储工具函数

### 3.1 创建向量存储工具

**文件**: `backend/app/agent/utils/vector_store.py`（新建）

```python
# backend/app/agent/utils/vector_store.py
"""
向量存储工具函数

提供业务知识向量库的创建和配置。
"""

import os
import logging
from langchain_chroma import Chroma
from langchain_community.embeddings import DashScopeEmbeddings

logger = logging.getLogger(__name__)


def create_business_vector_store(
    persist_directory: str = "./chroma_langchain_db",
    collection_name: str = "example_collection",
) -> Chroma:
    """
    创建业务知识向量库实例

    Args:
        persist_directory: 向量库持久化目录
        collection_name: 集合名称

    Returns:
        Chroma 向量库实例
    """
    # 获取 DashScope API Key
    dashscope_api_key = os.environ.get("DASHSCOPE_API_KEY")
    if not dashscope_api_key:
        raise ValueError(
            "DASHSCOPE_API_KEY 环境变量未设置，无法创建向量库"
        )

    # 初始化 Embedding 模型
    embeddings = DashScopeEmbeddings(model="text-embedding-v3")

    # 创建向量库实例
    vector_store = Chroma(
        collection_name=collection_name,
        embedding_function=embeddings,
        persist_directory=persist_directory,
    )

    logger.info(
        f"业务知识向量库已创建: {persist_directory}/{collection_name}"
    )

    return vector_store
```

```python
def _initialize_agent(self) -> None:
    """初始化 Agent"""
    try:
        # 配置代理设置
        _configure_proxy_settings()

        # 1. 创建 LLM
        llm = _create_llm(self._use_ollama)

        # 2. 连接数据库
        db, _ = _create_database_connection()

        # 3. 准备工具
        tools = _prepare_tools(db, llm)

        # 4. 构建系统提示词
        system_prompt = _build_system_prompt(db)

        # 5. 创建业务知识向量库
        try:
            vector_store = create_business_vector_store()
            rag_middleware = BusinessRagMiddleware(vector_store=vector_store)
            logger.info("业务知识 RAG 中间件已启用")
        except Exception as e:
            logger.warning(f"业务知识向量库初始化失败，RAG 功能将不可用: {e}")
            rag_middleware = None

        # 6. 配置 SummarizationMiddleware
        summarization_middleware = SummarizationMiddleware(
            model=llm,
            trigger=("tokens", 4000),
            keep=("messages", 5),
        )

        # 7. 构建中间件列表
        middleware_list = [
            summarization_middleware,
            SkillMiddleware(),
        ]
        if rag_middleware:
            middleware_list.insert(0, rag_middleware)  # RAG 中间件放在最前面

        # 8. 创建 Agent
        self.agent = create_agent(
            model=llm,
            tools=tools,
            system_prompt=system_prompt,
            middleware=middleware_list,
        )

        logger.info(
            "SQL Agent 初始化成功（SummarizationMiddleware、SkillMiddleware 和 BusinessRagMiddleware 已启用）"
        )

    except Exception as e:
        logger.error(f"SQL Agent 初始化失败: {e}")
        raise
```

```python
# backend/app/agent/middleware/test_rag_middleware.py
"""
BusinessRagMiddleware 单元测试
"""

import pytest
from langchain_core.messages import HumanMessage, SystemMessage
from langchain_core.documents import Document

from backend.app.agent.middleware.rag_middleware import BusinessRagMiddleware
from backend.app.agent.state import CustomState


class MockVectorStore:
    """模拟向量库"""
    def __init__(self, docs: list[Document]):
        self.docs = docs

    def similarity_search(self, query: str, k: int = 5):
        return self.docs[:k]


def test_before_model_with_human_message():
    """测试用户消息时触发检索"""
    # 准备测试数据
    docs = [
        Document(page_content="业务知识1", metadata={"title": "知识1"}),
        Document(page_content="业务知识2", metadata={"title": "知识2"}),
    ]
    vector_store = MockVectorStore(docs)
    middleware = BusinessRagMiddleware(vector_store=vector_store, top_k=2)

    # 准备 state
    state: CustomState = {
        "messages": [HumanMessage(content="用户查询")],
    }

    # 执行 before_model
    result = middleware.before_model(state)

    # 验证结果
    assert result is not None
    assert "messages" in result
    assert len(result["messages"]) == 2  # SystemMessage + HumanMessage
    assert isinstance(result["messages"][0], SystemMessage)
    assert "rag_context" in result
    assert "rag_query" in result
    assert result["rag_query"] == "用户查询"


def test_before_model_with_non_human_message():
    """测试非用户消息时跳过检索"""
    vector_store = MockVectorStore([])
    middleware = BusinessRagMiddleware(vector_store=vector_store)

    state: CustomState = {
        "messages": [SystemMessage(content="系统消息")],
    }

    result = middleware.before_model(state)

    # 应该返回 None，不执行检索
    assert result is None


def test_message_deduplication():
    """测试消息去重逻辑"""
    docs = [Document(page_content="业务知识", metadata={"title": "知识"})]
    vector_store = MockVectorStore(docs)
    middleware = BusinessRagMiddleware(vector_store=vector_store)

        "messages": [

            HumanMessage(content="新的查询"),

    result = middleware.before_model(state)

    # 验证旧消息被替换

    assert len(result["messages"]) == 2
    # 新的业务知识系统消息应该替换旧的
```

## 9. 注意事项

### 9.1 错误处理

- 向量库初始化失败时，应该优雅降级，不影响 Agent 的正常运行
- 检索失败时，应该记录日志但不中断流程

### 9.2 性能优化

- 考虑缓存常见查询的检索结果
- 限制检索数量，避免 Token 消耗过大
- 考虑异步检索（如果支持）

### 9.3 调试技巧

- 启用详细日志，查看检索过程和结果
- 检查 `rag_context` 和 `rag_query` 字段，验证检索是否正确执行
- 查看注入的系统消息内容，验证格式化是否正确

## 10. 常见问题

### Q: 为什么检索没有触发？

A: 检查以下几点：
1. 确保最后一条消息是用户消息（HumanMessage）
2. 确保向量库已正确初始化
3. 检查日志中是否有错误信息

### Q: 检索结果没有注入？

A: 检查以下几点：
1. 确保检索返回了文档（`retrieved_docs` 不为空）
2. 检查 `before_model` 的返回值是否正确
3. 查看日志确认中间件是否执行

### Q: 消息重复添加？

A: 检查消息去重逻辑：
1. 确保标识符 `__business_rag_context__` 正确使用
2. 检查 `_has_rag_system_message` 方法是否正确识别已有消息
